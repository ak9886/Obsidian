# How to set up a recursion function for a Mix-Max Algorithm

```Pseudocode
def minimax(state, depth, is_maximizing):

    # 1️⃣ Base Case
    if depth == 0 or is_terminal(state):
        return evaluate(state)

    # 2️⃣ Maximizing Player
    if is_maximizing:
        best_score = -infinity
        for child in generate_moves(state):
            score = minimax(child, depth - 1, False)
            best_score = max(best_score, score)
        return best_score

    # 3️⃣ Minimizing Player
    else:
        best_score = +infinity
        for child in generate_moves(state):
            score = minimax(child, depth - 1, True)
            best_score = min(best_score, score)
        return best_score
```

```Python
import math

# Board is a list of 9 cells
# X = AI (maximizing)
# O = Human (minimizing)
# " " = empty

def print_board(board):
    for i in range(0, 9, 3):
        print(board[i] + "|" + board[i+1] + "|" + board[i+2])
    print()


def is_winner(board, player):
    win_positions = [
        [0,1,2], [3,4,5], [6,7,8],   # rows
        [0,3,6], [1,4,7], [2,5,8],   # columns
        [0,4,8], [2,4,6]             # diagonals
    ]
    for pos in win_positions:
        if board[pos[0]] == board[pos[1]] == board[pos[2]] == player:
            return True
    return False


def is_terminal(board):
    return (
        is_winner(board, "X") or
        is_winner(board, "O") or
        " " not in board
    )


def evaluate(board):
    if is_winner(board, "X"):
        return 10
    elif is_winner(board, "O"):
        return -10
    else:
        return 0


def minimax(board, depth, is_maximizing):

    if is_terminal(board):
        return evaluate(board)

    if is_maximizing:
        best_score = -math.inf
        for i in range(9):
            if board[i] == " ":
                board[i] = "X"
                score = minimax(board, depth + 1, False)
                board[i] = " "
                best_score = max(best_score, score)
        return best_score

    else:
        best_score = math.inf
        for i in range(9):
            if board[i] == " ":
                board[i] = "O"
                score = minimax(board, depth + 1, True)
                board[i] = " "
                best_score = min(best_score, score)
        return best_score


def best_move(board):
    best_score = -math.inf
    move = None

    for i in range(9):
        if board[i] == " ":
            board[i] = "X"
            score = minimax(board, 0, False)
            board[i] = " "
            if score > best_score:
                best_score = score
                move = i

    return move


# --------- Play Example ---------

board = [" "] * 9

while not is_terminal(board):
    print_board(board)

    # Human move
    human_move = int(input("Enter position (0-8): "))
    if board[human_move] == " ":
        board[human_move] = "O"
    else:
        continue

    if is_terminal(board):
        break

    # AI move
    ai_move = best_move(board)
    board[ai_move] = "X"

print_board(board)

if is_winner(board, "X"):
    print("AI wins!")
elif is_winner(board, "O"):
    print("You win!")
else:
    print("Draw!")
```
