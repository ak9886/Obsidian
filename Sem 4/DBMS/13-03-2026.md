# Cursors
### Write a procedure using Cursor to display all orders with total amount.
```MySQL
DELIMITER $$

CREATE PROCEDURE DisplayAllRecords()
BEGIN
    DECLARE done INT DEFAULT 0;
    DECLARE v_id INT;
    DECLARE v_name VARCHAR(255);
    DECLARE v_age INT;
    DECLARE v_reason VARCHAR(255);
    DECLARE cur CURSOR FOR
        SELECT id, name, age, reason_for_visit FROM plsql;

    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = 1;

    OPEN cur;

    read_loop: LOOP
        FETCH cur INTO v_id, v_name, v_age, v_reason;

        IF done THEN
            LEAVE read_loop;
        END IF;

        SELECT 
            v_id AS ID,
            v_name AS Name,
            v_age AS Age,
            v_reason AS Reason_For_Visit;

    END LOOP;

    CLOSE cur;

END $$

DELIMITER ;
```
